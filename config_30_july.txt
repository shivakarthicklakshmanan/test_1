# selector when there's any such ambiguity, e.g.:
#     CONFIRM_BUTTON_TARGET = "input[value='Confirm']"

# Some dashboards take a long time to render (observed ~49s on this portal).

# Hard numeric safety net for the blind-dump walk (scrape_all_dashboards).
# visited_urls + MAX_DRILL_DEPTH already prevent a true infinite loop, but
# if the portal appends a session token/timestamp to URLs, the *same*
# logical page can look like a *different* URL every visit, defeating that
# check. This caps total pages visited across the whole run regardless,
# so a run always terminates in practice, not just in theory.
